# VTuber 化 P0 / P0.5 / P1 実装計画

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** アバター設定・モデル一覧 API・この PC 用プロファイルを整え（P0 / P0.5）、TTS 音声をブラウザ（WebAudio）で再生できるようにする（P1）。

**Architecture:** 設定は `schema.ts` を真実源に `[avatar]` / `[gui]` / `[tts]` / `llm` を拡張し、`config/settings.default.toml` を `npm run config:default` で同期する。音声は `PlaybackSink`（`core/ports.ts`）を新設し、`ServerPlaybackSink`（ffplay）と `WsPlaybackSink`（WS 送信 + クライアント ack 待ち）を `PlaybackRouter` で切り替える。`tts_audio` は `PipelineEvent` に載せ、`WsHub` が primary client にだけ配信する。

**Tech Stack:** TypeScript 5（strict）/ Node 24.11 / express + ws / esbuild（IIFE）/ `node:test` / VOICEVOX（WAV 24kHz）/ Ollama

## Global Constraints

- 作業ブランチは `develop`。コミットメッセージは日本語で「何を・なぜ」。
- 各タスクの最後に `npm run verify`（typecheck → test → build）を通す。テストは `node --test --test-timeout=30000`（`npm test`）。
- フロントは **esbuild の IIFE バンドル**（`dist/frontend/app.js`）。`tsc` 出力を配信しない。
- 設定の既定値・型は `src/backend/config/schema.ts` が単一の真実源。`config/settings.default.toml` は `npm run config:default` で同期（手で書かない）。
- 純粋ロジックは `src/shared` に置き `node:test` でテストする。I/O と分離する。
- `services → server` を import しない（依存は内向き）。`server → services` は可。
- `ClientCommand` に型を足したら `WsHub.commandHandlers` に実装する（未実装はコンパイルエラー）。
- コメント・ログ・UI 文言は日本語。`Logger` は `deps.logger ?? new Logger("Scope")`。
- 検証用の一時ファイル・スクリプトを残さない。最後に `git status` を確認する。
- 実測環境: Windows / RTX 3060 12GB / i7-6700 / RAM 40GB / Node v24.11.1 / Edge 157 / OBS 30 系 / Ollama `gemma4:e2b-it-qat` / VOICEVOX `speaker_id=2`。

`src/shared/avatar/modelCatalog.ts`:

```ts
/**
 * Live2D モデル（model3.json）の解析と URL 化（純粋ロジック）。
 * ファイル走査は `src/backend/services/avatarCatalog.ts` が担う。
 */

export interface AvatarModelInfo {
  /** プロジェクトルート基準のパス（`/` 区切り） */
  path: string;
  /** ブラウザから取得する URL */
  url: string;
  /** 表示名（フォルダ名） */
  label: string;
  /** モデルの表情名 */
  expressions: string[];
  /** モーションのグループ名 */
  motions: string[];
}

/** Windows の `\` を `/` に直し、先頭の `/` を落とす */
export function toUrlPath(path: string): string {
  return path.replace(/\\/g, "/").replace(/^\/+/, "");
}

/** model3.json から表情名とモーショングループ名を取り出す（不正なら null） */
export function parseModel3Json(raw: unknown): { expressions: string[]; motions: string[] } | null {
  if (!raw || typeof raw !== "object") {
    return null;
  }
  const refs = (raw as { FileReferences?: unknown }).FileReferences;
  if (!refs || typeof refs !== "object") {
    return null;
  }
  const { Expressions, Motions } = refs as { Expressions?: unknown; Motions?: unknown };
  const expressions = Array.isArray(Expressions)
    ? Expressions.map((entry) =>
        entry && typeof entry === "object" ? (entry as { Name?: unknown }).Name : undefined
      ).filter((name): name is string => typeof name === "string" && name.length > 0)
    : [];
  const motions =
    Motions && typeof Motions === "object" ? Object.keys(Motions as Record<string, unknown>) : [];

  // 表情もモーションも無い model3.json は「使えるモデル」とみなさない
  if (expressions.length === 0 && motions.length === 0) {
    return null;
  }
  return { expressions, motions };
}

/** 表示用のモデル情報を組み立てる */
export function createAvatarModelInfo(options: {
  relativePath: string;
  label: string;
  meta: { expressions: string[]; motions: string[] };
}): AvatarModelInfo {
  const path = toUrlPath(options.relativePath);
  return {
    path,
    url: `/${path}`,
    label: options.label,
    expressions: options.meta.expressions,
    motions: options.meta.motions,
  };
}
```

import * as fs from "fs";
import * as path from "path";
import {
  AvatarModelInfo,
  createAvatarModelInfo,
  parseModel3Json,
} from "../../shared/avatar/modelCatalog";

/**
 * `live2d-models/<dir>/*.model3.json` を列挙する（ディレクトリが無ければ空配列）。
 * 壊れた JSON でも候補には残す（画面で選べるようにするため）。
 */
export function listAvatarModels(rootDir: string): AvatarModelInfo[] {
  const baseDir = path.join(rootDir, "live2d-models");
  if (!fs.existsSync(baseDir)) {
    return [];
  }

  const models: AvatarModelInfo[] = [];
  for (const entry of fs.readdirSync(baseDir, { withFileTypes: true })) {
    if (!entry.isDirectory()) {
      continue;
    }
    const dir = path.join(baseDir, entry.name);
    for (const file of fs.readdirSync(dir)) {
      if (!file.toLowerCase().endsWith(".model3.json")) {
        continue;
      }
      const full = path.join(dir, file);
      let meta = { expressions: [] as string[], motions: [] as string[] };
      try {
        meta = parseModel3Json(JSON.parse(fs.readFileSync(full, "utf-8"))) ?? meta;
      } catch {
        // 解析できなくても候補としては返す
      }
      models.push(
        createAvatarModelInfo({
          relativePath: path.relative(rootDir, full),
          label: entry.name,
          meta,
        })
      );
    }
  }

  return models.sort((a, b) => a.label.localeCompare(b.label));
}

`src/shared/audio/envelope.ts`:

```ts
import { wavToFloat32 } from "./pcm";

/**
 * WAV（16bit PCM）から `sliceMs` ごとの RMS を求め、最大値で正規化する（0..1）。
 *
 * Open-LLM-VTuber と同じ方式。クライアントは再生経過時間からこの配列を引くだけで
 * 口パクを駆動できる（AnalyserNode に依存しない）。無音・解析不能なら空配列。
 */
export function buildVolumeEnvelope(wav: Uint8Array, sliceMs = 20): number[] {
  const decoded = wavToFloat32(wav);
  if (!decoded || decoded.samples.length === 0) {
    return [];
  }

  const sliceSamples = Math.max(1, Math.round((decoded.sampleRate * sliceMs) / 1000));
  const volumes: number[] = [];
  for (let offset = 0; offset < decoded.samples.length; offset += sliceSamples) {
    const end = Math.min(offset + sliceSamples, decoded.samples.length);
    let sum = 0;
    for (let i = offset; i < end; i += 1) {
      sum += decoded.samples[i] * decoded.samples[i];
    }
    volumes.push(Math.sqrt(sum / (end - offset)));
  }

  const max = volumes.reduce((acc, value) => Math.max(acc, value), 0);
  if (max === 0) {
    return [];
  }
  return volumes.map((value) => Math.min(1, value / max));
}
```

/** クライアントの役割。display は音声を鳴らす担当（OBS / デスクトップ窓） */
export type ClientRole = "display" | "control";

export interface ClientIdentity {
  id: string;
  role: ClientRole;
}

/** `?role=display` を読む（既定は control） */
export function parseClientRole(url: string | undefined | null): ClientRole {
  if (!url) {
    return "control";
  }
  const index = url.indexOf("?");
  if (index < 0) {
    return "control";
  }
  const params = new URLSearchParams(url.slice(index + 1));
  return params.get("role") === "display" ? "display" : "control";
}

/** primary（音声を鳴らすクライアント）を選ぶ。display 優先、無ければ先頭 */
export function pickPrimaryClientId(clients: ClientIdentity[]): string | null {
  return clients.find((client) => client.role === "display")?.id ?? clients[0]?.id ?? null;
}

```ts
import { bytesToBase64 } from "../../shared/audio/pcm";
import type { ServerMessage } from "../../shared/protocol";
import { Logger } from "../core/logger";
import type { PlaybackHandle, PlaybackMeta, PlaybackSink } from "../core/ports";

export interface WsPlaybackSinkDeps {
  /** 配信（`tts_audio` は WsHub が primary に絞る） */
  send: (message: ServerMessage) => boolean;
  /** 音声を鳴らせるクライアントがいるか */
  hasPrimaryClient: () => boolean;
  logger?: Logger;
  /** ack が来ないときの猶予（ms） */
  timeoutMarginMs?: number;
}

/**
 * ブラウザ（WebAudio）で再生する sink。
 * `tts_audio` を送り、クライアントの `audio_playback_done` を待つ。
 * 通知が来なくても **必ず** `durationMs + 猶予` で解決する（`speaking` で固まらないため）。
 */
export class WsPlaybackSink implements PlaybackSink {
  readonly kind = "browser" as const;
  private readonly logger: Logger;
  private readonly pending = new Map<string, () => void>();

  constructor(private readonly deps: WsPlaybackSinkDeps) {
    this.logger = deps.logger ?? new Logger("Playback");
  }

  isAvailable(): boolean {
    return this.deps.hasPrimaryClient();
  }

  play(wav: Buffer, meta: PlaybackMeta): PlaybackHandle {
    if (!this.isAvailable()) {
      return { finished: Promise.resolve(), stop: () => undefined };
    }

    let resolveFinished!: () => void;
    const finished = new Promise<void>((resolve) => {
      resolveFinished = resolve;
    });
    let settled = false;
    let timer: ReturnType<typeof setTimeout>;
    const settle = (): void => {
      if (settled) {
        return;
      }
      settled = true;
      this.pending.delete(meta.utteranceId);
      clearTimeout(timer);
      resolveFinished();
    };

    timer = setTimeout(() => {
      this.logger.warn(`再生完了通知が来ません: ${meta.utteranceId}（タイムアウトで継続します）`);
      settle();
    }, Math.max(0, meta.durationMs) + (this.deps.timeoutMarginMs ?? 1500));
    (timer as unknown as { unref?: () => void }).unref?.();
    this.pending.set(meta.utteranceId, settle);

    const sent = this.deps.send({
      type: "event",
      event: {
        type: "tts_audio",
        utteranceId: meta.utteranceId,
        text: meta.text,
        audio: bytesToBase64(wav),
        volumes: meta.volumes,
        sliceMs: meta.sliceMs,
        durationMs: meta.durationMs,
      },
    });
    if (!sent) {
      this.logger.warn("音声を送信できませんでした（クライアント未接続）");
      settle();
    }

    return { finished, stop: () => settle() };
  }

  /** クライアントの `audio_playback_done` を受ける */
  resolveDone(utteranceId: string): void {
    this.pending.get(utteranceId)?.();
  }

  /** 待機中の再生をすべて解く（終了処理用） */
  dispose(): void {
    for (const settle of [...this.pending.values()]) {
      settle();
    }
  }
}
```

 * ブラウザ再生を指定されていても表示クライアントが居なければサーバー再生へ落とす。

```ts
import type { ClientCommand, HelloPayload, PipelineEvent } from "../shared/protocol";
import { bytesFromBase64 } from "../shared/audio/pcm";

type AudioEvent = Extract<PipelineEvent, { type: "tts_audio" }>;

export interface AudioPlayerDeps {
  send(command: ClientCommand): void;
  onPlayingChange?(playing: boolean): void;
}

/**
 * `tts_audio` を WebAudio で順番に再生する。
 * 再生が終わったら `audio_playback_done` を返し、サーバーの `speaking` を終わらせる。
 */
export class AudioPlayer {
  private ctx: AudioContext | null = null;
  private gain: GainNode | null = null;
  private volume = 1;
  private readonly ready: { event: AudioEvent; buffer: AudioBuffer }[] = [];
  private current: AudioBufferSourceNode | null = null;
  private playing = false;
  /** stop 後に古い onended が ack を送らないようにするための世代番号 */
  private token = 0;

  constructor(private readonly deps: AudioPlayerDeps) {}

  configure(cfg: HelloPayload["audio"]): void {
    this.volume = Math.min(1, Math.max(0, cfg.outputVolume));
    if (this.gain) {
      this.gain.gain.value = this.volume;
    }
  }

  /** ユーザー操作の中で呼ぶ（自動再生ポリシー対策） */
  async prime(): Promise<void> {
    const ctx = this.context();
    if (ctx.state === "suspended") {
      try {
        await ctx.resume();
      } catch {
        // 次の操作で再試行する
      }
    }
  }

  enqueue(event: AudioEvent): void {
    void this.decode(event);
  }

  stop(): void {
    this.token += 1;
    this.ready.length = 0;
    if (this.current) {
      try {
        this.current.stop();
      } catch {
        // 既に停止している場合は無視
      }
      this.current = null;
    }
    if (this.playing) {
      this.playing = false;
      this.deps.onPlayingChange?.(false);
    }
  }

  dispose(): void {
    this.stop();
    void this.ctx?.close().catch(() => undefined);
    this.ctx = null;
    this.gain = null;
  }

  private context(): AudioContext {
    if (!this.ctx) {
      this.ctx = new AudioContext();
      this.gain = this.ctx.createGain();
      this.gain.gain.value = this.volume;
      this.gain.connect(this.ctx.destination);
    }
    return this.ctx;
  }

  private async decode(event: AudioEvent): Promise<void> {
    try {
      const copy = new Uint8Array(bytesFromBase64(event.audio));
      const buffer = await this.context().decodeAudioData(copy.buffer as ArrayBuffer);
      this.ready.push({ event, buffer });
      this.pump();
    } catch (error) {
      console.warn("[Audio] 音声をデコードできませんでした", error);
      this.deps.send({ type: "audio_playback_done", utteranceId: event.utteranceId });
    }
  }

  private pump(): void {
    if (this.playing || this.ready.length === 0) {
      return;
    }
    const next = this.ready.shift()!;
    const source = this.context().createBufferSource();
    source.buffer = next.buffer;
    if (this.gain) {
      source.connect(this.gain);
    }

    const token = this.token;
    this.playing = true;
    this.current = source;
    this.deps.onPlayingChange?.(true);

    source.onended = () => {
      if (token !== this.token) {
        return;
      }
      this.current = null;
      this.playing = false;
      this.deps.send({ type: "audio_playback_done", utteranceId: next.event.utteranceId });
      this.pump();
    };

    source.start();
  }
}
```
